// Die Bedien-Schicht in der Vorschau (W3c-1; Vorlage: Entwurf w3c-v2) – läuft IM abgeschotteten iframe (sandbox="allow-scripts",
// kein allow-same-origin) und nirgends sonst. Sie meldet der Erweiterung nur per postMessage, was passiert ist: Zahlen für die
// Stelle (Abschnitt i, Spalte j, Baustein k in der Reihenfolge der Seite; beim Ziehen die Stelle, VOR der etwas landet) und – nur beim Text direkt tippen – den getippten
// Text. Mehr nimmt die Erweiterung nicht an (`nachricht.ts`), und Links prüft sie mit der echten Regel.
//
// `config` kommt von der Erweiterung: { scroll, symbole, modus: 'computer' | 'handy', darfHtml, abstaende, abstandNamen, aufteilungen, weiter }.
// Maße ziehen (W3c-3): Spaltenbreite (`aufteilung`) und Abstand (`abstand`) – auch hier nur Zahlen bzw. ein Name aus den Listen der Erweiterung.
// `K` ist der Kern (`ebene.kern.txt`): Kollisionsregel und Textmodell – das, was sich ohne Seite prüfen lässt.
(function (config, K) {
  'use strict';
  const S = config.symbole;
  const senden = (m) => { parent.postMessage(Object.assign({ websiteEditor: true }, m), '*'); };
  const haupt = document.getElementById('inhalt');
  // Am Handy gibt es kein Ziehen und kein Tippen im Text: nur Abschnitt-Leiste, „Abschnitt einfügen“ und Tippen auf einen
  // Baustein öffnet das Fenster (wie bisher).
  const voll = config.modus !== 'handy';
  const arr = (x) => Array.prototype.slice.call(x);
  const icon = (p) => S[p] || '';

  const abschnitte = () => (haupt ? arr(haupt.children).filter((e) => e.matches('section.w-abschnitt')) : []);
  const innenVon = (sec) => sec.querySelector(':scope > .w-abschnitt-innen');
  const spaltenVon = (sec) => { const i = innenVon(sec); return i ? arr(i.children).filter((e) => e.classList.contains('w-spalte')) : []; };
  const bausteineVon = (sp) => arr(sp.children).filter((e) => e.classList.contains('w-baustein'));
  const ortVon = (block) => {
    const sp = block.parentElement;
    const sec = sp.closest('section.w-abschnitt');
    return { i: abschnitte().indexOf(sec), j: spaltenVon(sec).indexOf(sp), k: bausteineVon(sp).indexOf(block) };
  };
  const seiteX = () => window.scrollX;
  const seiteY = () => window.scrollY;
  const artVon = (b) => (b.className.match(/w-b-([a-zA-Z]+)/) || [])[1];

  // ── Bausteine ──────────────────────────────────────────────────────────────────────────────────────────────
  const knopf = (was, symbol, text, titel) => {
    const b = document.createElement('button');
    b.type = 'button';
    b.dataset.w = was;
    if (titel) { b.title = titel; b.setAttribute('aria-label', titel); }
    b.innerHTML = icon(symbol) + (text ? '<span>' + text + '</span>' : '');
    return b;
  };
  const abstandVon = (sec) => {
    for (let n = 0; n < config.abstaende.length; n++) if (sec.classList.contains('w-abschnitt--abstand-' + config.abstaende[n])) return config.abstaende[n];
    return 'normal';
  };

  const abstandText = (sec) => 'Abstand oben & unten: ' + (config.abstandNamen[abstandVon(sec)] || '');

  // ── Die Schicht ────────────────────────────────────────────────────────────────────────────────────────────
  // ALLE Bedienelemente (Leisten, Griff, „+“, Rahmen) liegen in EINER Schicht über der Seite und werden nur bei Bedarf gezeigt –
  // immer an einer Stelle, an der sie weder Text noch ein anderes Bedienelement berühren (`K.ordne`, siehe `layout`).
  const ueber = document.createElement('div');
  ueber.className = 'b-ueber';
  document.body.appendChild(ueber);
  const neu = (tag, klasse) => { const d = document.createElement(tag); d.className = klasse; d.hidden = true; ueber.appendChild(d); return d; };
  const rahmenSec = neu('div', 'b-rahmen b-rahmen--abschnitt');
  const rahmenBlock = neu('div', 'b-rahmen b-rahmen--baustein');

  // Abschnitt-Leiste: Griff (ziehen oder Pfeiltasten), Gestalten, Kopieren, Löschen
  const werk = neu('div', 'b-werk');
  let werkFuer = null;
  const griff = knopf('griff', 'griff', '', voll ? 'Abschnitt verschieben: ziehen oder Pfeil hoch/runter' : 'Abschnitt verschieben: Pfeil hoch/runter');
  griff.classList.add('b-griff-abschnitt');
  werk.appendChild(griff);
  // Am Handy gibt es kein Ziehen: dort bleiben „nach oben“ und „nach unten“ (am Computer ersetzt der Griff sie).
  (voll ? [] : [['hoch', 'hoch', '', 'Abschnitt nach oben'], ['runter', 'runter', '', 'Abschnitt nach unten']]).concat([['gestalten', 'stift', 'Gestalten', 'Abschnitt gestalten'], ['kopieren', 'kopie', '', 'Abschnitt kopieren'], ['loeschen', 'muell', '', 'Abschnitt löschen']]).forEach((a) => {
    const k = knopf(a[0], a[1], a[2], a[3]);
    if (a[0] === 'loeschen') k.classList.add('b-gefahr');
    werk.appendChild(k);
  });
  werk.addEventListener('click', (e) => {
    const k = e.target.closest('button');
    if (!k || k.dataset.w === 'griff' || !werkFuer) return;
    e.stopPropagation();
    const i = abschnitte().indexOf(werkFuer);
    if (k.dataset.w === 'hoch' || k.dataset.w === 'runter') { // eine Stelle weiter; am Rand nichts
      const nach = k.dataset.w === 'hoch' ? i - 1 : i + 2;
      if (nach >= 0 && nach <= abschnitte().length) senden({ was: 'verschieben', art: 'abschnitt', von: i, nach: nach });
      return;
    }
    senden({ was: k.dataset.w, i: i });
  });
  // Pfeiltasten am Griff: eine Stelle weiter. „Stelle“ heißt: vor welchem Abschnitt (0 = ganz oben, Anzahl = ganz unten).
  griff.addEventListener('keydown', (e) => {
    if ((e.key !== 'ArrowUp' && e.key !== 'ArrowDown') || !werkFuer) return;
    e.preventDefault();
    const i = abschnitte().indexOf(werkFuer);
    const nach = e.key === 'ArrowUp' ? i - 1 : i + 2;
    if (nach < 0 || nach > abschnitte().length) return;
    senden({ was: 'verschieben', art: 'abschnitt', von: i, nach: nach });
  });

  // Baustein-Leiste (EINE helle Leiste: Stift = das Fenster des Bausteins) – wandert zum Baustein unter der Maus
  let blockleiste = null;
  let blockFuer = null;
  let griffBaustein = null;
  if (voll) {
    blockleiste = neu('div', 'b-blockleiste');
    griffBaustein = knopf('griff', 'griff', '', 'Baustein verschieben: ziehen oder Pfeil hoch/runter');
    griffBaustein.classList.add('b-griff-baustein');
    blockleiste.appendChild(griffBaustein);
    blockleiste.appendChild(knopf('fenster', 'stift', '', 'Baustein im Fenster bearbeiten'));
    blockleiste.addEventListener('click', (e) => {
      const k = e.target.closest('button');
      if (!k || k.dataset.w === 'griff' || !blockFuer) return;
      e.preventDefault();
      e.stopPropagation();
      const o = ortVon(blockFuer);
      senden({ was: 'baustein', i: o.i, j: o.j, k: o.k });
    });
  }

  // „+ Abschnitt einfügen“ an der Grenze zwischen zwei Abschnitten (nur wenn die Maus dicht an der Grenze ist)
  const einfBtn = knopf('einfuegen', 'plus', 'Abschnitt einfügen');
  einfBtn.classList.add('b-einfuegen');
  einfBtn.hidden = true;
  ueber.appendChild(einfBtn);
  let einfAn = 0;
  einfBtn.onclick = (e) => { e.stopPropagation(); senden({ was: 'einfuegen', i: einfAn }); };

  // Abstand-Pille unten mittig (W3c-3): senkrecht ziehen rastet auf die Stufen ein, ein Klick ohne Ziehen öffnet „Gestalten“.
  // Das Modell hat EINEN Abstand für oben und unten – die Pille und der Chip sagen das. Pfeil hoch/runter = eine Stufe.
  let abstandBtn = null;
  let abstandFuer = null;
  const baender = [];
  if (voll) {
    abstandBtn = document.createElement('button');
    abstandBtn.type = 'button';
    abstandBtn.className = 'b-abstand';
    abstandBtn.hidden = true;
    abstandBtn.title = 'Abstand oben und unten: ziehen. Ein Klick öffnet „Gestalten“.';
    abstandBtn.setAttribute('aria-label', 'Abstand oben und unten ändern: ziehen, Pfeil hoch oder runter, oder Enter für „Gestalten“');
    abstandBtn.innerHTML = icon('regler') + '<span></span>';
    ueber.appendChild(abstandBtn);
    ['oben', 'unten'].forEach((seite) => { const d = neu('div', 'b-band b-band--' + seite); baender.push(d); });
  }

  // „+“ in der Lücke zwischen zwei Bausteinen (öffnet das Hinzufügen)
  const plus = neu('button', 'b-plus');
  plus.type = 'button';
  plus.innerHTML = icon('plus');
  plus.title = 'Baustein hier hinzufügen';
  plus.setAttribute('aria-label', 'Baustein hier hinzufügen');
  const plusLinie = neu('div', 'b-pluslinie');
  let plusOrt = null;
  plus.addEventListener('click', (e) => { e.stopPropagation(); if (plusOrt) senden({ was: 'baustein-neu', i: plusOrt.i, j: plusOrt.j }); });

  // Spalten-Griffe (W3c-3): einer je Lücke zwischen zwei Spalten, mittig, sichtbar am Abschnitt unter der Maus (`layout`). Ziehen geht,
  // wenn es für die Spaltenzahl mindestens zwei Aufteilungen gibt (heute: zwei Spalten) – sonst ist der Griff gedämpft und sagt, warum.
  const aufteilungVon = (sec) => { const m = innenVon(sec).className.match(/w-spalten--(\d(?:-\d)*)/); return m ? m[1] : '1'; };
  const teilerListe = [];
  if (voll) {
    abschnitte().forEach((sec) => {
      const n = spaltenVon(sec).length;
      if (n < 2) return;
      const namen = config.aufteilungen[String(n)] || [];
      const frei = K.aufteilungFrei(namen);
      for (let j = 0; j < n - 1; j++) {
        const t = neu('button', 'b-teiler');
        t.type = 'button';
        t.innerHTML = '<i></i>';
        if (frei) {
          t.title = 'Spaltenbreite ziehen (oder Pfeil links/rechts)';
          t.setAttribute('aria-label', 'Spaltenbreite ändern: ziehen oder Pfeil links und rechts');
        } else {
          t.classList.add('b-fest');
          t.title = 'Bei ' + n + ' Spalten gibt es nur eine Aufteilung – die Spaltenzahl änderst du über „Gestalten“.';
          t.setAttribute('aria-label', t.title);
          t.setAttribute('aria-disabled', 'true');
        }
        teilerListe.push({ el: t, sec: sec, j: j, n: n, namen: namen, frei: frei });
      }
    });
  }

  // Leere Spalten: ein Knopf in der Spalte
  abschnitte().forEach((sec) => {
    spaltenVon(sec).forEach((sp) => {
      const bl = bausteineVon(sp);
      bl.forEach((b) => {
        if (!voll) { // Handy: Tippen öffnet das Fenster
          b.addEventListener('click', (e) => { e.preventDefault(); e.stopPropagation(); const o = ortVon(b); senden({ was: 'baustein', i: o.i, j: o.j, k: o.k }); });
          return;
        }
        // Klick auf den Baustein selbst: Text → direkt tippen (siehe unten), alles andere das Fenster
        b.addEventListener('click', (e) => {
          if (e.target.closest('.b-ueber')) return;
          e.preventDefault();
          if (!textBaustein(b) && !(artVon(b) === 'html' && !config.darfHtml)) { e.stopPropagation(); const o = ortVon(b); senden({ was: 'baustein', i: o.i, j: o.j, k: o.k }); }
        });
      });
      if (bl.length === 0) {
        const n = document.createElement('button');
        n.type = 'button';
        n.className = 'b-leerspalte';
        n.innerHTML = icon('plus') + '<span>Baustein hinzufügen</span>';
        n.onclick = (e) => { e.stopPropagation(); senden({ was: 'baustein-neu', i: abschnitte().indexOf(sec), j: spaltenVon(sec).indexOf(sp) }); };
        sp.appendChild(n);
      }
    });
  });
  if (haupt && abschnitte().length === 0) {
    const leer = document.createElement('div');
    leer.className = 'b-leer';
    const p = document.createElement('p');
    p.textContent = 'Diese Seite ist noch leer.';
    const b0 = knopf('einfuegen', 'plus', 'Abschnitt einfügen');
    b0.onclick = () => { senden({ was: 'einfuegen', i: 0 }); };
    leer.appendChild(p);
    leer.appendChild(b0);
    haupt.appendChild(leer);
  }

  // In der Vorschau führt nichts weg und nichts wird abgeschickt.
  document.addEventListener('click', (e) => { const a = e.target.closest && e.target.closest('a'); if (a) e.preventDefault(); }, true);
  document.addEventListener('submit', (e) => { e.preventDefault(); }, true);

  // ── Text direkt tippen ─────────────────────────────────────────────────────────────────────────────────────
  // Welche Felder sich tippen lassen: 'plain' = ein Feld mit reinem Text (Überschrift, Akzentzeile, Knopf), 'md' = Text mit
  // Auszeichnung (Text, Text neben dem Bild). „Eigenes HTML“ und alles andere nie.
  const FELD_WAHL = 'h1,h2,h3,.w-akzent,.w-text,.w-knopf';
  const feldDaten = (el) => {
    const block = el.closest('.w-baustein');
    if (!block) return null;
    const typ = artVon(block);
    if (typ === 'ueberschrift' && el.matches('h1,h2,h3') && el.parentElement === block) return { block: block, feld: 'text', art: 'plain' };
    if (typ === 'akzentzeile' && el.matches('.w-akzent')) return { block: block, feld: 'text', art: 'plain' };
    if ((typ === 'text' || typ === 'bildText') && el.matches('.w-text')) return { block: block, feld: 'markdown', art: 'md' };
    if (typ === 'knoepfe' && el.matches('.w-knopf')) return { block: block, feld: 'knopf', art: 'plain', n: arr(block.querySelectorAll('.w-knopf')).indexOf(el) };
    return null;
  };
  const textBaustein = (b) => voll && /w-b-(ueberschrift|akzentzeile|text|bildText|knoepfe)\b/.test(b.className) && !!b.querySelector(FELD_WAHL);
  // Das Feld unter einem Klickziel (auch ein Zeichen in einem Link im Text; eine Zwischenüberschrift IM Text gehört zum Textfeld)
  const feldUnter = (ziel) => {
    let el = ziel && ziel.closest && ziel.closest(FELD_WAHL);
    while (el) {
      const d = feldDaten(el);
      if (d) return { el: el, d: d };
      el = el.parentElement && el.parentElement.closest(FELD_WAHL);
    }
    return null;
  };

  let aktiv = null; // { el, d, original, verlauf }
  let letzterDruck = { ziel: null, zeit: 0 };
  const sel = () => getSelection();
  const reinText = (el) => (el.textContent || '').replace(/ /g, ' ').replace(/\s+/g, ' ').trim();

  // Eigener Rückgängig-Verlauf für Text mit Auszeichnung (Änderungen am DOM durch die Leiste zerstören den des Browsers).
  // Ein Stand ist der Text (Blöcke) mit der Auswahl dazu; ob sich etwas geändert hat, entscheidet NUR der Text – sonst
  // zerstörte schon eine andere Auswahl den Weg zum Wiederholen.
  const momentaufnahme = () => {
    const r = K.lesen(aktiv.el, sel());
    return { b: r.bloecke, a: r.auswahl, j: JSON.stringify(r.bloecke) };
  };
  const merke = (hart) => {
    const v = aktiv && aktiv.verlauf;
    if (!v) return;
    const s = momentaufnahme();
    const jetzt = Date.now();
    if (v.stapel[v.pos].j === s.j) { v.stapel[v.pos].a = s.a; return; }
    if (!hart && v.pos > 0 && jetzt - v.zeit < 800) v.stapel[v.pos] = s;
    else { v.stapel.length = v.pos + 1; v.stapel.push(s); v.pos++; }
    v.zeit = jetzt;
  };
  const zeige = (s) => {
    K.rendern(aktiv.el, s.b);
    if (s.a) K.auswahlSetzen(aktiv.el, s.a);
    formatLeiste.aktualisieren();
  };
  const rueckgaengig = (wiederholen) => {
    const v = aktiv && aktiv.verlauf;
    if (!v) return;
    merke(true); // was seit dem letzten Stand getippt wurde, nicht verlieren (gleicher Text → es ändert sich nichts)
    const n = v.pos + (wiederholen ? 1 : -1);
    if (n < 0 || n >= v.stapel.length) return;
    v.pos = n;
    zeige(v.stapel[n]);
  };
  /** Einen Rechenschritt des Kerns auf den Text im Feld anwenden (`auswahl` aus dem DOM, wenn nicht gegeben). */
  const anwenden = (schritt, auswahlGemerkt) => {
    if (!aktiv || aktiv.d.art !== 'md') return;
    merke(true);
    const r = K.lesen(aktiv.el, auswahlGemerkt ? null : sel());
    const a = auswahlGemerkt || r.auswahl;
    if (!a) return;
    const neu2 = schritt(r.bloecke, a);
    K.rendern(aktiv.el, neu2.bloecke);
    aktiv.el.focus();
    K.auswahlSetzen(aktiv.el, neu2.auswahl);
    merke(true);
    formatLeiste.aktualisieren();
  };

  const vorbereiten = (el, d) => {
    if (d.art === 'plain') {
      const punkt = el.querySelector('.w-punkt'); // der Punkt hinter der Überschrift gehört nicht zum Text
      if (punkt) punkt.remove();
      el.contentEditable = 'plaintext-only';
      if (el.contentEditable !== 'plaintext-only') el.contentEditable = 'true';
    } else {
      const r = K.lesen(el, null); // gleiche Gestalt wie später beim Lesen: p, h3, ul/ol – und ein leerer Text bekommt einen Absatz
      K.rendern(el, r.bloecke);
      el.contentEditable = 'true';
    }
    el.spellcheck = true;
    el.classList.add('b-edit');
  };
  const caretAn = (el, x, y) => {
    let r = null;
    if (x != null) {
      if (document.caretRangeFromPoint) r = document.caretRangeFromPoint(x, y);
      else if (document.caretPositionFromPoint) { const cp = document.caretPositionFromPoint(x, y); if (cp) { r = document.createRange(); r.setStart(cp.offsetNode, cp.offset); r.collapse(true); } }
    }
    if (!r || !el.contains(r.startContainer)) { r = document.createRange(); r.selectNodeContents(el); r.collapse(false); }
    const s = sel();
    s.removeAllRanges();
    s.addRange(r);
  };
  const stelleDes = (d) => { const o = ortVon(d.block); return { i: o.i, j: o.j, k: o.k, feld: d.feld, n: d.n == null ? 0 : d.n }; };

  const bearbeiten = (feld, x, y) => {
    if (aktiv) beenden('uebernehmen', null);
    const el = feld.el;
    const d = feld.d;
    aktiv = { el: el, d: d, original: el.innerHTML, verlauf: null };
    vorbereiten(el, d);
    el.focus();
    caretAn(el, x, y);
    if (d.art === 'md') aktiv.verlauf = { stapel: [momentaufnahme()], pos: 0, zeit: 0 };
    document.body.classList.add('b-tippt');
    senden(Object.assign({ was: 'tippen' }, stelleDes(d)));
  };

  // Eine Bearbeitung beenden. art: 'uebernehmen' | 'abbrechen'. weiter: das nächste Feld (Klick auf ein anderes Textfeld)
  const beenden = (art, weiter) => {
    if (!aktiv) return;
    const a = aktiv;
    aktiv = null;
    const el = a.el;
    const d = a.d;
    document.body.classList.remove('b-tippt');
    formatLeiste.zu();
    el.removeAttribute('contenteditable');
    el.classList.remove('b-edit');
    sel().removeAllRanges();
    planeLayout();
    const fertig = () => { senden({ was: 'tippen-ende' }); if (weiter) bearbeiten(weiter, weiter.x, weiter.y); };
    if (art === 'abbrechen') { el.innerHTML = a.original; senden({ was: 'tippen-ende' }); return; }
    const probe = document.createElement(el.tagName);
    probe.innerHTML = a.original;
    const punktProbe = probe.querySelector('.w-punkt');
    if (punktProbe) punktProbe.remove();
    const wert = d.art === 'md' ? K.fuerSpeichern(K.lesen(el, null).bloecke) : reinText(el);
    const alt = d.art === 'md' ? K.fuerSpeichern(K.lesen(probe, null).bloecke) : reinText(probe);
    if (JSON.stringify(wert) === JSON.stringify(alt)) { el.innerHTML = a.original; fertig(); return; } // unverändert: alles wie vorher
    const msg = Object.assign({ was: 'text' }, stelleDes(d));
    if (d.art === 'md') msg.bloecke = wert; else msg.wert = wert;
    if (weiter) { const w = stelleDes(weiter.d); msg.weiter = { i: w.i, j: w.j, k: w.k, feld: w.feld, n: w.n }; }
    senden(msg);
    senden({ was: 'tippen-ende' });
  };

  // Klick auf ein Textfeld → tippen. Läuft in der Aufnahmephase, damit Links im Text nicht springen.
  document.addEventListener('pointerdown', (e) => { letzterDruck = { ziel: e.target, zeit: Date.now(), x: e.clientX, y: e.clientY }; }, true);
  document.addEventListener('click', (e) => {
    if (!voll) return;
    if (e.target.closest('.b-format, .b-ueber, .b-leerspalte')) return;
    const f = feldUnter(e.target);
    if (!f) return;
    e.preventDefault();
    e.stopPropagation();
    if (aktiv && aktiv.el === f.el) return; // schon dabei: den Cursor setzt der Browser
    bearbeiten(f, e.clientX, e.clientY);
  }, true);
  // Klick daneben übernimmt (auch Klick in ein anderes Textfeld: dann dort gleich weiter)
  document.addEventListener('focusout', (e) => {
    if (!aktiv || e.target !== aktiv.el) return;
    setTimeout(() => {
      if (!aktiv) return;
      const im = document.activeElement;
      if (im && im.closest && im.closest('.b-format')) return; // die Leiste (Link-Feld) – nicht beenden
      if (im === aktiv.el) return;
      let weiter = null;
      if (letzterDruck.ziel && Date.now() - letzterDruck.zeit < 400) {
        const f = feldUnter(letzterDruck.ziel);
        if (f && f.el !== aktiv.el && !letzterDruck.ziel.closest('.b-ueber')) weiter = { el: f.el, d: f.d, x: letzterDruck.x, y: letzterDruck.y };
      }
      beenden('uebernehmen', weiter);
    }, 0);
  });
  document.addEventListener('keydown', (e) => {
    if (!aktiv) return;
    if (e.key === 'Escape') { e.preventDefault(); beenden('abbrechen'); return; }
    if (e.key === 'Enter' && aktiv.d.art === 'plain') { e.preventDefault(); beenden('uebernehmen'); return; }
    if (e.key === 'Enter' && aktiv.d.art === 'md' && (e.metaKey || e.ctrlKey)) { e.preventDefault(); beenden('uebernehmen'); return; }
    // Rückgängig/Wiederholen im Text mit Auszeichnung über den EIGENEN Verlauf. Über die Tasten, nicht nur über `beforeinput`:
    // Der Browser meldet „Wiederholen“ nur, wenn in SEINEM Verlauf etwas zum Wiederholen liegt – tut es nach unserem
    // Rückgängig nie (gemessen 21.11.2026).

  // des Feldes). Die Tasten kommen beim Rahmen sonst nicht an, solange der Fokus im iframe liegt.
  document.addEventListener('keydown', (e) => {
    if (aktiv || zieh || (!e.metaKey && !e.ctrlKey) || e.altKey) return;
    const k = e.key.toLowerCase();
    const wieder = (k === 'z' && e.shiftKey) || (k === 'y' && !e.metaKey);
    if (k !== 'z' && !wieder) return;
    e.preventDefault();
    senden({ was: wieder ? 'wiederholen' : 'rueckgaengig' });
  });

  // ── Scroll merken, Start ───────────────────────────────────────────────────────────────────────────────────
  addEventListener('scroll', () => { senden({ was: 'scroll', y: Math.round(window.scrollY) }); }, { passive: true });
  if (config.scroll > 0) window.scrollTo(0, config.scroll);

  if (config.weiter && voll) { // direkt im nächsten Feld weitertippen (Klick während der Bearbeitung eines anderen Feldes)
    const w = config.weiter;
    const sec = abschnitte()[w.i];
    const col = sec && spaltenVon(sec)[w.j];
    const blk = col && bausteineVon(col)[w.k];
    let el = null;
    if (blk) {
      if (w.feld === 'knopf') el = blk.querySelectorAll('.w-knopf')[w.n];
      else if (w.feld === 'markdown') el = blk.querySelector('.w-text');
      else el = blk.querySelector('h1,h2,h3,.w-akzent');
    }
    const f = el && feldUnter(el);
    if (f) bearbeiten(f, null, null);
  }
  senden({ was: 'bereit' });
})
